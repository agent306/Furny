# 03 — Design Documents (Plan & Furniture JSON)

Read the master document, `01`, and `02` first. This file defines the **exact document contracts** shared by the editors, the engine, the server and the vision service. Types below are normative TypeScript; the engine implements each as a zod schema with the same name suffixed `Schema` (e.g. `PlanDocumentSchema`) in `engine/plan/schema.ts` and `engine/furniture/schema.ts`, and exports `type X = z.infer<typeof XSchema>`. The Python vision service mirrors `PlanDocument` with pydantic models in `vision/app/schemas/plan.py` (generated by hand; a contract test in `16` validates vision output against the zod schema via the engine CLI).

All lengths: integer mm (`Mm = number` refined `.int()`). All IDs: see master §6.3.

---

## 1. Plan document

```ts
type Mm = number;            // integer millimetres
type Deg = number;           // degrees [0, 360)
type Point = { x: Mm; y: Mm };

interface PlanDocument {
  schemaVersion: 1;
  meta: PlanMeta;
  nodes: Record<string, WallNode>;          // id → node
  walls: Record<string, Wall>;
  openings: Record<string, Opening>;
  rooms: Record<string, RoomMeta>;          // metadata for detected rooms (see §1.5)
  placements: Record<string, FurniturePlacement>;
  fixtures: Record<string, Fixture>;
  annotations: {
    dimensions: Record<string, DimensionAnnotation>;
    labels: Record<string, TextLabel>;
  };
  background: BackgroundImage | null;
  view: ViewState;                          // last camera/view, not semantically important
}

interface PlanMeta {
  defaultWallHeight: Mm;          // e.g. 2700
  defaultWallThickness: Mm;       // interior default, e.g. 115
  floorThickness: Mm;             // slab drawn in 3D, default 150
  northAngle: Deg | null;         // optional, for display only
  createdWith: 'editor' | 'import' | 'trace';
}

interface WallNode {
  id: string;                     // n_xxxxxxxxxx
  x: Mm; y: Mm;
}

interface Wall {
  id: string;                     // w_…
  a: string;                      // node id
  b: string;                      // node id (a !== b)
  thickness: Mm;                  // 50..1000
  height: Mm;                     // 300..10000
  kind: 'exterior' | 'interior' | 'partition' | 'half';  // 'half' = low wall/parapet (height typically ≤ 1200)
  justification: 'center';        // v1 only centre-line walls (offset walls are a phase-2 option)
  material: 'brick' | 'concrete' | 'drywall' | 'glass' | 'other';
  colorHex: string | null;        // interior paint colour for 3D, null = default
}

interface Opening {
  id: string;                     // o_…
  wallId: string;
  kind: 'door' | 'window' | 'opening';     // 'opening' = empty arch/passage
  type: DoorType | WindowType | 'opening';
  offset: Mm;                     // distance along wall from node a to the opening CENTRE, measured on the centre line
  width: Mm;                      // clear width along the wall
  height: Mm;
  sillHeight: Mm;                 // 0 for doors/openings
  hinge: 'start' | 'end';         // doors: hinge at the end nearer node a ('start') or node b ('end')
  swing: 'left' | 'right';        // doors: side of the wall the leaf swings into, relative to direction a→b ('left' = +normal, see 04 §2.1)
  label: string | null;           // e.g. "D1", "W2"
}
type DoorType = 'single_swing' | 'double_swing' | 'sliding' | 'pocket' | 'bifold' | 'main_entrance';
type WindowType = 'casement' | 'sliding_window' | 'fixed' | 'french_window' | 'bay' | 'ventilator';

interface RoomMeta {
  id: string;                     // r_…
  anchor: Point;                  // a point strictly inside the room face; used to re-associate after geometry changes (04 §4.4)
  name: string;                   // "Master Bedroom"
  type: RoomType;
  floorFinish: 'tile' | 'wood' | 'marble' | 'carpet' | 'concrete' | 'other';
  floorColorHex: string | null;
  labelOffset: Point | null;      // manual label position relative to polygon centroid; null = auto
  showArea: boolean;
}
type RoomType = 'living' | 'dining' | 'kitchen' | 'bedroom' | 'master_bedroom' | 'bathroom' | 'toilet' | 'balcony'
  | 'utility' | 'store' | 'study' | 'pooja' | 'foyer' | 'corridor' | 'dress' | 'deck' | 'ots' | 'garage' | 'other';

interface FurniturePlacement {
  id: string;                     // f_…
  designId: string;               // furniture_designs.id (ULID)
  x: Mm; y: Mm;                   // centre of BACK edge of footprint (master §6.2)
  rotation: Deg;                  // clockwise, 0 = front faces +y
  elevation: Mm;                  // bottom of furniture above floor (0 for floor-standing; wall cabinets e.g. 1400)
  mirrored: boolean;              // mirror across local x (left/right swap) — affects rendering only, NOT the BOM; v1: hinge sides visually mirrored
  wallAttachment: { wallId: string; side: 'left' | 'right' } | null; // set when snapped flush to a wall face
}

// At rotation = 0 a fixture's front faces +y, the same convention as placements (master §6.2); (x, y) is the footprint centre.
interface Fixture {
  id: string;                     // x_…
  libraryKey: FixtureKey;         // static library in 05 §9 (bed_double, sofa_3, wc, basin, kitchen_counter, fridge, …)
  x: Mm; y: Mm;                   // centre of footprint
  rotation: Deg;
  width: Mm; depth: Mm; height: Mm;
  elevation: Mm;
  label: string | null;
}

interface DimensionAnnotation {
  id: string;                     // d_…
  a: Point; b: Point;             // measured points
  offset: Mm;                     // perpendicular offset of the dimension line (signed)
  attachedTo: { kind: 'wall'; wallId: string; face: 'left' | 'right' | 'center' } | null; // if attached, a/b are recomputed from the wall
}

interface TextLabel {
  id: string;                     // t_…
  position: Point;
  text: string;                   // ≤ 200 chars
  fontSize: Mm;                   // in plan space (scales with zoom), default 250
  rotation: Deg;
}

interface BackgroundImage {
  importId: string | null;        // blueprint_imports.id when from an import
  mediaUuid: string;              // medialibrary uuid of the image (normalized PNG)
  pixelWidth: number; pixelHeight: number;
  mmPerPixel: number;             // float allowed; scale
  position: Point;                // plan-space coordinate of the image's top-left pixel
  rotation: Deg;
  opacity: number;                // 0..1, default 0.5
  visible: boolean;
  locked: boolean;
}

interface ViewState {
  mode2d: { centerX: Mm; centerY: Mm; zoom: number } | null;    // zoom = screen px per mm
  mode3d: { camera: 'orbit' | 'top' | 'walk'; position: [number, number, number]; target: [number, number, number] } | null; // metres
  lastMode: '2d' | '3d';
}
```

### 1.1 Plan invariants (validated by `validatePlan`; errors block saving, warnings do not)

| Code | Severity | Rule |
|---|---|---|
| `plan.node_missing` | error | Every `wall.a`/`wall.b` references an existing node |
| `plan.wall_degenerate` | error | Wall length ≥ 10 mm |
| `plan.wall_duplicate` | error | No two walls share the same unordered node pair |
| `plan.orphan_node` | warning (auto-fixed on save by removing) | Node not referenced by any wall |
| `plan.opening_wall_missing` | error | `opening.wallId` exists |
| `plan.opening_out_of_bounds` | error | `width/2 ≤ offset ≤ wallLength − width/2` |
| `plan.opening_overlap` | error | Openings on the same wall don't overlap (≥ 0 mm gap) |
| `plan.opening_too_tall` | error | `sillHeight + height ≤ wall.height` |
| `plan.placement_design_missing` | warning | `designId` refers to a design in the same project (checked server-side only) |
| `plan.walls_intersect` | warning | Two walls cross without sharing a node (editor auto-splits on creation; can exist after import) |
| `plan.room_anchor_outside` | warning | RoomMeta anchor not inside any face (room metadata orphaned) |
| `plan.too_large` | error | > 2000 walls, > 1000 openings, > 500 placements, > 1000 fixtures, or JSON > 2 MB |
| numeric ranges | error | thickness 50–1000, height 300–10000, opening width 300–6000, sill 0–3000, coordinates within ±1 000 000 mm |

### 1.2 Derived plan geometry (not stored; computed by `derivePlanGeometry(doc)` in the engine)

Derived coordinates are floats (`number`), not integer mm; `areaMm2` and `perimeterMm` are rounded integers. `derivePlanGeometry` returns `DerivedPlan` (04 §7.1), a structural superset of `PlanGeometry`.

```ts
interface PlanGeometry {
  walls: Record<string, {
    centerline: [Point, Point];
    outline: Point[];               // mitered polygon (CW), see 04 §3
    leftFace: [Point, Point]; rightFace: [Point, Point];
    length: Mm;                     // centre-line length
    openings: Array<{ id: string; u0: Mm; u1: Mm }>; // along-wall interval of each opening
  }>;
  rooms: Array<{
    faceKey: string;                // stable hash of sorted wall ids bounding the face
    polygon: Point[];               // inner (net) polygon = face minus the union of wall outlines (04 §4.5)
    holes: Point[][];               // CCW hole rings of the net polygon; [] when none
    areaMm2: number;
    perimeterMm: Mm;
    centroid: Point;
    metaId: string | null;          // RoomMeta associated via anchor
  }>;
  bbox: { min: Point; max: Point };
  exteriorOutline: Point[] | null;  // union of all wall outlines' outer boundary
}
```

## 2. Furniture document

```ts
interface FurnitureDocument {
  schemaVersion: 1;
  templateKey: TemplateKey;               // see 07 §2
  family: 'carcass' | 'table';
  dims: { width: Mm; height: Mm; depth: Mm };   // overall OUTER bounding box incl. plinth/legs/top overhangs
  materials: MaterialAssignments;
  carcass: CarcassSpec | null;            // required iff family = 'carcass'
  table: TableSpec | null;                // required iff family = 'table'
  overrides: Overrides;
  notes: string | null;                   // customer/staff notes shown in BOM/production sheet
}

type MaterialRole = 'carcass' | 'back' | 'shelf' | 'front' | 'top' | 'drawer_box' | 'drawer_bottom' | 'plinth' | 'leg';

type MaterialAssignments = Record<MaterialRole, {
  materialCode: string;                   // materials.code
  edgeBandCode: string | null;            // edge band for visible edges of this role; null = role default (07 §5)
  finishCode: string | null;              // finishes.code; null = none
}>;
```

### 2.1 Carcass family

```ts
interface CarcassSpec {
  construction: {
    top: 'between_sides' | 'over_sides';
    bottom: 'between_sides' | 'over_sides';
  };
  mounting: 'floor' | 'wall' | 'loft';     // wall/loft: no plinth, hanging rails/brackets added
  plinth: { type: 'none' | 'plinth' | 'legs'; height: Mm; setback: Mm; legCode: string | null };
  back: { type: 'none' | 'nailed' | 'grooved' | 'inset' };
  countertop: { enabled: boolean; overhangFront: Mm; overhangLeft: Mm; overhangRight: Mm } | null; // separate top panel in 'top' material
  interior: Section;                       // root section = full interior box
  fronts: Front[];
  handles: { code: string | null; orientation: 'auto' | 'vertical' | 'horizontal'; placement: 'auto' | 'top' | 'middle' | 'bottom' };
}

interface Section {
  id: string;                              // s_…
  split: 'none' | 'columns' | 'rows';      // columns = vertical partitions (left→right), rows = fixed shelves (top→bottom)
  children: SectionChild[];                // length ≥ 2 iff split ≠ 'none'; [] otherwise
  content: Content | null;                 // non-null iff split = 'none'
}

interface SectionChild {
  size: { mode: 'fixed'; mm: Mm } | { mode: 'flex'; weight: number };   // flex shares remaining space by weight (> 0)
  section: Section;
}

type Content =
  | { type: 'empty' }
  | { type: 'shelves'; count: number; adjustable: boolean; frontSetback: Mm }                // count 1..20, evenly spaced
  | { type: 'drawers'; count: number; frontHeights: 'equal' | Mm[]; slideCode: string | null; behindDoor: boolean } // count 1..10
  | { type: 'hanging_rod'; topClearance: Mm; rodCode: string | null }                         // rod centre this far below section top (default 60)
  | { type: 'pull_out_basket'; count: number; basketCode: string | null };                   // kitchen pull-outs (hardware only)

interface Front {
  id: string;                              // fr_…
  kind: 'hinged' | 'sliding' | 'flap_up' | 'flap_down';
  covers: string[];                        // section ids; their union must be an axis-aligned rectangle (validated)
  leaves: number | 'auto';                 // hinged: number of leaves side by side; auto = ceil(width / maxDoorWidth) (07 §4)
  hingeSide: 'left' | 'right' | 'auto';    // single-leaf hinged only; auto = right
  overlay: 'full' | 'half' | 'inset';
  glass: boolean;                          // glass panel insert (front role becomes frame + glass, 07 §4.5)
  handleCode: string | null | 'inherit';   // 'inherit' = carcass.handles.code
  hingeCode: string | null | 'auto';
  slidingTrackCode: string | null | 'auto';// sliding only
}
```

Drawer fronts are **not** `Front`s: they are generated from `drawers` content (unless `behindDoor: true`, in which case drawers get no visible front, only internal drawer boxes with a plain front panel in `drawer_box` material).

```ts
  storage: {
```

### 2.3 Overrides (staff and advanced customer edits that survive regeneration)

```ts
interface Overrides {
  parts: Record<PartKey, PartOverride>;
  extraParts: ExtraPart[];
  hardware: Record<HardwareKey, HardwareOverride>;
  extraHardware: ExtraHardware[];
}

type PartKey = string;      // deterministic, see §3.6
type HardwareKey = string;  // deterministic, see §3.6

interface PartOverride {
  removed?: true;
  materialCode?: string;
  edges?: Partial<Record<EdgeId, string | null>>;   // edge band code per edge, null = no banding
  lengthDelta?: Mm;                                  // ± adjustment to generated finished length
  widthDelta?: Mm;
  grain?: 'length' | 'width' | 'none';
  note?: string;
}

interface ExtraPart {
  id: string;                    // p_…
  name: string;
  role: MaterialRole;
  materialCode: string;
  length: Mm; width: Mm;         // finished size
  qty: number;                   // 1..99
  edges: Record<EdgeId, string | null>;
  grain: 'length' | 'width' | 'none';
  note: string | null;
  placement: { min: [Mm, Mm, Mm]; size: [Mm, Mm, Mm] } | null;   // optional box in furniture local space for 3D display
}

interface HardwareOverride { removed?: true; code?: string; qtyDelta?: number; note?: string }
interface ExtraHardware { id: string; code: string; qty: number; note: string | null }   // id h_…

type EdgeId = 'L1' | 'L2' | 'W1' | 'W2';   // L1/L2 = the two long edges (along length), W1/W2 = the two short edges
```

### 2.4 Furniture invariants (validated by `validateFurniture`)

Detailed rules (including construction warnings like shelf sag) are in `07` §7. Hard errors that block saving (= `SAVE_BLOCKING_CODES`, 07 §7.1; the engine CLI `validate-furniture` returns `valid = !issues.some(i => SAVE_BLOCKING_CODES.has(i.code))`). Every other `error` issue of 07 §7.3 does **not** block autosave but blocks order submission (09 T2) and proposal sending (09 T6):

| Code | Rule |
|---|---|
| `furniture.dims_range` | width 150–6000, height 100–3000, depth 100–1200 |
| `furniture.family_mismatch` | `carcass` non-null iff family carcass; `table` non-null iff family table |
| `furniture.section_tree_invalid` | split/children/content consistency; ids unique |
| `furniture.section_too_small` | any leaf section < 100 mm wide or < 80 mm high (net) |
| `furniture.fixed_sizes_exceed` | sum of fixed child sizes + dividers > available space |
| `furniture.front_not_rectangle` | front `covers` union is not a rectangle |
| `furniture.front_overlap` | two fronts cover the same section |
| `furniture.unknown_code` | any material/edge band/hardware/finish code not in catalog snapshot (active **or** archived — archived codes produce warning `furniture.archived_code`) |
| `furniture.too_many_parts` | > 400 generated parts |

## 3. Derived furniture model, BOM and price (engine outputs)

### 3.1 `FurnitureModel` — `generateFurniture(doc, catalog)`

```ts
interface FurnitureModel {
  parts: Part[];
  hardware: HardwareLine[];
  fronts: Array<{ id: string; frontId: string | null; kind: 'door' | 'drawer' | 'flap' | 'sliding'; rect: { x: Mm; y: Mm; w: Mm; h: Mm }; hingeSide?: 'left' | 'right'; partKey: PartKey }>; // for 2D elevation and 3D animation
  sections: Array<{ id: string; box: Box3 }>;     // interior box per leaf section (for UI picking and dimensioning)
  bbox: Box3;
  issues: Issue[];
}

interface Box3 { min: [Mm, Mm, Mm]; size: [Mm, Mm, Mm] }   // furniture local space (master §6.2)

interface Part {
  key: PartKey;
  nameKey: string;                       // i18n key, e.g. 'parts.side_left'
  nameParams: Record<string, string | number>;
  role: MaterialRole;
  materialCode: string;
  thickness: Mm;                         // rounded from material thickness_mm (e.g. 18)
  length: Mm;                            // finished size along the part's length axis (≥ width unless grain forces otherwise)
  width: Mm;
  grain: 'length' | 'width' | 'none';    // which finished dimension must follow sheet grain
  edges: Record<EdgeId, string | null>;  // edge band codes
  qty: number;                           // usually 1; identical parts are NOT merged here (merging happens in BOM)
  geometry: { kind: 'box'; box: Box3 } | { kind: 'extruded'; outline: Array<[Mm, Mm]>; plane: 'xz'; y: Mm; thickness: Mm }; // for rendering
  operations: { hingeCups: number; shelfPinHoles: number; slideScrewSets: number; handleHoles: number; dowels: number; camLocks: number };
  visibleFaces: number;                  // 0, 1 or 2 — used for finish area
  source: 'generated' | 'extra';
  overridden: boolean;
}

interface HardwareLine {
  key: HardwareKey;
  code: string;                          // hardware_items.code
  qty: number;
  reasonKey: string;                     // i18n, e.g. 'hardware.reason.hinges_for_door'
  reasonParams: Record<string, string | number>;
  frontId: string | null;
  sectionId: string | null;
  source: 'generated' | 'extra';
  overridden: boolean;
}

interface Issue {
  code: string;                          // e.g. 'furniture.shelf_sag'
  severity: 'error' | 'warning' | 'info';
  messageKey: string;                    // i18n key
  params: Record<string, string | number>;
  entityIds: string[];                   // section/front/part keys involved
}
```

### 3.2 `Bom` — `computeBom(model, catalog, options)`

`options: { nesting: boolean; allowRotation?: boolean; finishByRole: Record<MaterialRole, string | null> }`, built by `bomOptionsFor(doc, nesting)` (08 §3).

```ts
interface Bom {
  engineVersion: string;
  catalogVersion: number;
  cutList: CutListRow[];                 // identical parts merged
  materials: Array<{
    materialCode: string;
    partCount: number;
    netAreaMm2: number;                  // sum of finished part areas × qty
    sheetsNested: number | null;         // from nesting (null if nesting disabled)
    sheetsEstimated: number;             // ceil(netArea × (1 + waste_factor) / sheetArea) — always present
    sheetsBilled: number;                // = sheetsNested ?? sheetsEstimated
    utilization: number | null;          // 0..1 from nesting
  }>;
  edgeBanding: Array<{ edgeBandCode: string; lengthMm: number }>;  // includes the 50 mm per-edge allowance AND 5 % waste (08 §3.3)
  hardware: Array<{ code: string; qty: number }>;                  // merged, after overrides
  finishing: Array<{ finishCode: string; areaMm2: number }>;
  operations: { hingeCups: number; shelfPinHoles: number; dowels: number; camLocks: number; slideScrewSets: number; handleHoles: number };
  nesting: NestingResult | null;
  issues: Issue[];
}

interface CutListRow {
  rowKey: string;                        // hash(material, thickness, length, width, grain, edges)
  partKeys: PartKey[];
  name: string;                          // resolved English name for exports (UI re-translates via nameKey of first part)
  nameKey: string; nameParams: Record<string, string | number>;
  materialCode: string; thickness: Mm;
  length: Mm; width: Mm;                 // finished
  cutLength: Mm; cutWidth: Mm;           // finished − banding thickness (+ pre-mill allowance), see 08 §3.2
  qty: number;
  grain: 'length' | 'width' | 'none';
  edges: Record<EdgeId, string | null>;
}

interface NestingResult {
  sheets: Array<{
    materialCode: string;
    sheetLength: Mm; sheetWidth: Mm;
    placements: Array<{ rowKey: string; partKey: PartKey; x: Mm; y: Mm; length: Mm; width: Mm; rotated: boolean }>; // partKey = instance key: `key` for the first instance, `key#k` for k ≥ 2
    utilization: number;
    cuts: Array<{ x1: Mm; y1: Mm; x2: Mm; y2: Mm }>;   // guillotine cut lines for drawing
  }>;
  kerf: Mm; trim: Mm;
  unplaced: PartKey[];                   // parts larger than the sheet → issue 'bom.part_exceeds_sheet'
}
```

### 3.3 `PriceBreakdown` — `priceBom(bom, catalog, pricing, options)`

```ts
interface PriceBreakdown {

    description: string;                  // English, for exports
    quantity: number;                     // sheets, metres (2 dp), pieces, m² (2 dp), labour units
    unit: string;

  }>;
  materialCostMinor: number;
  edgeBandCostMinor: number;
  hardwareCostMinor: number;
  finishingCostMinor: number;
  labourCostMinor: number;
  directCostMinor: number;                // sum of the five above
```

### 3.5 Revisions & BOM authority

- Client computes `FurnitureModel`, `Bom` and `PriceBreakdown` live (Web Worker) for display. These are **estimates** and are stored only as `furniture_designs.estimate` summary.
- On every **furniture revision** creation, the server dispatches `ComputeRevisionBom` (queue `engine`), which runs `compute-bom` with nesting via the engine CLI against the current catalog and stores `bom`, `price`, `engine_version`, `catalog_version` on the revision. Quotes are built **only** from server-computed revision BOMs (`08` §6).

### 3.6 Deterministic part & hardware keys

Keys are dot-paths derived from document structure so that overrides keep pointing at the same physical part after unrelated edits:

| Part | Key pattern |
|---|---|
| Carcass sides | `carcass.side.left`, `carcass.side.right` |
| Carcass top/bottom | `carcass.top`, `carcass.bottom` |
| Back panel | `carcass.back` (or `carcass.back.{n}` when split for size > sheet) |
| Glass frame door parts of leaf i of front F | `front.{F}.leaf.{i}.stile.left`, `.stile.right`, `.rail.top`, `.rail.bottom`, `.glass` |
| Plinth front/back/returns | `plinth.front`, `plinth.back`, `plinth.return.{n}` |
| Countertop | `countertop` |

## 4. Schema versioning & migrations

- Every document carries `schemaVersion`. The engine exports `migratePlan(doc: unknown): PlanDocument` and `migrateFurniture(doc: unknown): FurnitureDocument` that upgrade any older version step-by-step (`v1→v2→…`). v1 is the only version at launch; the migration framework, its registry and a no-op test must exist.
- The server always stores the **migrated** document (engine CLI `validate-*` returns `migratedDocument`).
- Revisions are never rewritten; they are migrated on read.

## 5. Working copy, autosave and concurrency protocol

Applies identically to plans (`/api/plans/{plan}/working-copy`), furniture designs (`/api/furniture-designs/{design}/working-copy`) and staff proposal drafts (`/api/change-proposal-items/{item}/draft`).

**Client**
1. Editor loads `{ document, version }` from Inertia props (`workingDocument`, `workingVersion`).
2. Every applied command marks the store dirty and persists the document to `localStorage['fs:draft:{type}:{id}']` together with `baseVersion` (throttled 1 s).
3. Autosave: debounce 1500 ms after the last command, max wait 10 s, single in-flight request; a second change while saving queues one follow-up save.
4. Request: `PUT …/working-copy` JSON `{ document, baseVersion }` (CSRF header from Inertia/axios-free XHR helper `lib/http.ts`).
5. Responses:
   - `200 { version, savedAt, summary? , estimate? }` → set `baseVersion = version`, clear localStorage draft, status "Saved".
   - `409 { code: 'version_conflict', currentVersion, savedBy: {id, name}, savedAt }` → editor stops autosaving and shows the conflict dialog: **Reload theirs** (discard local), **Keep mine** (re-send with `baseVersion = currentVersion`, overwriting), **Save mine as copy** (plans: duplicate plan; furniture: duplicate design).
   - `423 { code: 'locked', reason: 'in_order', orderNumber }` → furniture design locked by an order; editor switches to read-only and shows banner with "Duplicate to edit".
   - `423 { code: 'locked', reason: 'plan_locked', lockedBy: {id, name}, lockedAt }` → plan locked by staff (02 §3.2 `plans.locked_at`); editor switches to read-only reason `plan_locked` (05 §15.3).
   - `422 { code: 'invalid_document', issues: Issue[] }` → should not happen (client validates first); show error toast, keep local draft, stop autosave, report via `console.error` + `POST /api/client-errors` (route `api.client-errors.store`; forwarded to Sentry when enabled, 17 §13.5).
   - network/5xx → retry with backoff 2, 4, 8, 16, 30 s (cap), status "Offline — changes kept on this device".
6. On page load, if `localStorage` has a draft with `baseVersion === workingVersion` and content differs, offer **Restore unsaved changes**. If `baseVersion < workingVersion`, offer restore as a conflict (same dialog as 409).
7. `beforeunload` warns when dirty and unsaved; a final `fetch(…, { keepalive: true, method: 'PUT' })` is attempted.

**Server**
1. Authorise (policy `update`), check lock: furniture → `locked_by_order_id` null (staff change locked designs only through the proposal draft endpoint, which requires `orders.propose_changes`, 13 §8.7); plans → `locked_at` null, else `423 { code: 'locked', reason: 'plan_locked', lockedBy, lockedAt }`.
2. Validate size (≤ 2 MB) then run engine CLI `validate-plan` / `validate-furniture` (furniture with catalog snapshot). Save-blocking errors (plans: every `error` issue of §1.1; furniture: `SAVE_BLOCKING_CODES` of §2.4) → 422.
3. `UPDATE … SET working_document = ?, working_version = working_version + 1, working_saved_by = ?, working_saved_at = now() WHERE id = ? AND working_version = ?` — 0 rows → 409.
4. Plans: recompute and store `summary` via `plan-summary` (same CLI process call batch: implement `validate-plan` to also return summary to avoid a second spawn). Furniture: update denormalised `width_mm/height_mm/depth_mm` from `validate-furniture` result (`bbox`) and `estimate` from the request header-less body field `estimate` sent by the client (display-only, trusted only for display).
5. If this is the first save by a **staff** user on a customer's plan since the last revision, create a `staff_edit` revision of the previous document first.
6. If no revision exists for the plan created today (business TZ), create an `auto_daily` revision of the **new** document (plans only).
7. Broadcast nothing (single editor assumption); log activity only for staff saves.

**Revisions API**: `POST …/revisions { label? }` → creates `manual_checkpoint` from the current working copy (furniture: also dispatches `ComputeRevisionBom`). `POST …/revisions/{rev}/restore` → creates `before_restore` revision of the current working copy, then copies the revision document into the working copy (version + 1).
